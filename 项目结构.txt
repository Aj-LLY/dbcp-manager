项目进度管理系统/                  版本: 4.5.1
│
├── main.py                          # 程序入口，初始化路径并启动主窗口
├── build_exe.py                     # 打包脚本，自动读取版本号生成带版本信息的EXE
├── release.py                       # 发布脚本，构建EXE并可自动创建GitHub Release
│
├── models/                          # 数据实体层 —— 定义核心数据结构
│   ├── __init__.py                  #   包初始化
│   ├── project.py                   #   项目实体（公司/系统/证书编号/下证日期/等级/属地/文件夹路径/交付日期/备注/阶段）
│   ├── workflow.py                  #   流程阶段实体（名称、排序、颜色、列宽）
│   ├── log_entry.py                 #   操作日志实体（7种操作类型常量、时间戳、关联项目、阶段变更）
│   └── dto.py                       #   数据传输对象（ProjectCreateDTO/UpdateDTO/ResultDTO/WorkflowStageDTO）
│
├── services/                        # 业务服务层 —— 封装数据操作逻辑
│   ├── __init__.py                  #   包初始化
│   ├── interfaces.py                #   5个抽象接口（IDataService/IProjectService/IWorkflowService/ILogService/IOleEmbedService）
│   ├── data_service.py              #   JSON数据持久化（单例、原子写入、_FileSerializer加密隔离、reload热加载）
│   ├── project_service.py           #   项目CRUD业务（创建、更新、删除、阶段移动、日志回调）
│   ├── workflow_service.py          #   流程配置管理（增删改查、排序、重置、列宽、replace_all_stages孤儿迁移）
│   ├── log_service.py               #   日志追踪服务（记录、查询、回调工厂、网络信息采集）
│   ├── backup_service.py            #   WebDAV备份服务（HTTP Basic认证，PUT/GET/PROPFIND/DELETE，GMT→CST时区）
│   ├── ole_service.py               #   OLE对象嵌入服务（Win32Com实现，原则#5技术隔离）
│   └── cert_ocr.py                  #   备案证OCR识别服务（easyocr+PyMuPDF，单例模式，关键词提取）
│
├── ui/                              # 用户界面层 —— Tkinter桌面GUI组件
│   ├── __init__.py                  #   包初始化
│   ├── main_window.py               #   主窗口控制器（MVC协调层，初始化服务、绑定事件、项目文件夹创建）
│   ├── toolbar.py                   #   顶部工具栏（新增/编辑流程/删除/刷新/备份/日志按钮）
│   ├── kanban_board.py              #   看板主组件（列容器、横向/纵向滚动、列高约束、卡片选中管理）
│   ├── kanban_column.py             #   看板列组件（阶段标题+计数、卡片列表、滚轮翻页、拖拽调整列宽）
│   ├── project_card.py              #   项目卡片（状态色条、左右箭头、详情/编辑/复制/文件夹/重命名/打包/报告打印按钮）
│   ├── project_dialog.py            #   新增/编辑项目对话框（公司名/系统名/等级/证书编号/下证日期/属地/交付日期/文件夹）
│   ├── workflow_dialog.py           #   流程配置对话框（增删改查、排序、颜色、列宽、重置默认）
│   ├── detail_dialog.py             #   项目详情对话框（信息展示、阶段移动、编辑/删除、日志摘要）
│   ├── log_dialog.py                #   操作日志查看器（表格展示、按项目筛选、水平滚动）
│   ├── backup_dialog.py             #   WebDAV备份对话框（服务器配置、备份/恢复/删除、自动刷新）
│   └── calendar_picker.py           #   日历日期选择器（月份/年份翻页、今天高亮、点击选日期）
│
├── utils/                           # 工具模块 —— 基础设施和通用函数
│   ├── __init__.py                  #   包初始化
│   ├── config.py                    #   全局配置（版本号、动态默认流程、窗口几何持久化、本地备份、UI常量）
│   ├── helpers.py                   #   通用辅助函数（ID生成、日期计算、名称验证、证书编号校验、边框输入框）
│   ├── logger.py                    #   操作日志记录器（内存+文件双写、增删查）
│   ├── crypto_utils.py              #   数据加密/解密工具
│   ├── error_log.py                 #   全局异常捕获模块
│   ├── tooltip.py                   #   悬浮提示组件
│   ├── province_data.py             #   中国省市区数据
│   └── webdav_config.py             #   WebDAV配置管理（服务器地址/用户名/密码、JSON持久化）
│
└── data/                            # 运行时数据目录（程序自动生成，不纳入版本控制）
    ├── dap_data.json                 #   主数据文件（项目列表 + 流程阶段配置）
    ├── operation_log.json            #   操作日志文件（所有增删改操作的记录）
    ├── webdav_config.json            #   WebDAV连接配置文件
    └── report_defaults.json          #   报告打印默认值配置
